<script>
fetch('data.json')
  .then(res => res.json())
  .then(data => {
    const list = document.getElementById('list');

    // show only first 2 items
    data.slice(0, 2).forEach(item => {
      const li = document.createElement('li');
      li.innerHTML = `<a href="${item.mylink}">${item.title}</a>`;
      list.appendChild(li);
    });
  })
  .catch(err => console.error(err));
</script>
